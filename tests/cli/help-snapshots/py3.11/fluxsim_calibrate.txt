usage: fluxsim calibrate [-h] [--playback-id PLAYBACK_ID] [--dry-run]
                         [--write] [--run-id RUN_ID] [--state-dir STATE_DIR]
                         [-j]
                         bundle

Read a Playback's declared scorecard and report the credibility and grade it
carries. Declared calibration only: no fitting is performed.

examples:
  # print the declared scorecard; nothing is written
  fluxsim calibrate ./bundle

  # record it in the run directory
  fluxsim calibrate ./bundle --write --run-id de72fa4b22711c52

arguments:
  bundle                path to the FLUX bundle directory

options:
  -h, --help            show this help message and exit
  --playback-id PLAYBACK_ID
                        default: the bundle's only Playback
  --dry-run             print the scorecard without writing calibration-
                        provenance.json (default)
  --write               write calibration-provenance.json into the run
                        directory
  --run-id RUN_ID       required without --dry-run
  --state-dir STATE_DIR
                        run-state directory (default: ./.flux)
  -j, --json            emit machine-readable JSON

docs: https://agenticstiger.github.io/flux/engine/reference/cli/calibrate.html
